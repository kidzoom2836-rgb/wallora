   100→export function ProjectsPanel() {
   101→  const { open, setOpen, selectedProjectId, setSelectedProjectId } = useProjects();
   102→  const openSignIn = useAuthDialog((s) => s.openSignIn);
   103→  const { status } = useSession();
   104→
   105→  // If the user signs out while the panel is open, close it.
   106→  React.useEffect(() => {
   107→    if (open && status === "unauthenticated") {
   108→      setOpen(false);
   109→    }
   110→  }, [open, status, setOpen]);
   111→
   112→  // Esc to close.
   113→  React.useEffect(() => {
   114→    if (!open) return;
   115→    const onKey = (e: KeyboardEvent) => {
   116→      if (e.key === "Escape") setOpen(false);
   117→    };
   118→    window.addEventListener("keydown", onKey);
   119→    return () => window.removeEventListener("keydown", onKey);
   120→  }, [open, setOpen]);
   121→
   122→  // Lock scroll while open.
   123→  React.useEffect(() => {
   124→    if (!open) return;
   125→    const prev = document.body.style.overflow;
   126→    document.body.style.overflow = "hidden";
   127→    return () => {
   128→      document.body.style.overflow = prev;
   129→    };
   130→  }, [open]);
   131→
   132→  // If not authenticated, surface sign-in instead of the panel.
   133→  const handleOpenChange = (next: boolean) => {
   134→    if (next && status !== "authenticated") {
   135→      openSignIn();
   136→      return;
   137→    }
   138→    setOpen(next);
   139→    if (!next) {
   140→      // Clear selection on close so reopening shows the sidebar / empty state.
   141→      setSelectedProjectId(null);
   142→    }
   143→  };
   144→
   145→  return (
   146→    <AnimatePresence>
   147→      {open && status === "authenticated" && (
   148→        <ProjectsPanelInner
   149→          onClose={() => handleOpenChange(false)}
   150→          selectedProjectId={selectedProjectId}
   151→          setSelectedProjectId={setSelectedProjectId}
   152→        />
   153→      )}
   154→    </AnimatePresence>
   155→  );
   156→}
   157→
   158→function ProjectsPanelInner({
   159→  onClose,
   160→  selectedProjectId,
   161→  setSelectedProjectId,
   162→}: {
   163→  onClose: () => void;
   164→  selectedProjectId: string | null;
   165→  setSelectedProjectId: (id: string | null) => void;
   166→}) {
   167→  const queryClient = useQueryClient();
   168→  const [showNewProjectForm, setShowNewProjectForm] = React.useState(false);
   169→  const [mobileSidebarOpen, setMobileSidebarOpen] = React.useState(false);
   170→
   171→  // Reset the "new project" form when the panel closes/opens.
   172→  React.useEffect(() => {
   173→    if (!selectedProjectId) setShowNewProjectForm(false);
   174→  }, [selectedProjectId]);
   175→
   176→  // ------------------------- Queries -------------------------
   177→  const { data: projectsData, isLoading: loadingList } = useQuery<{ data: ProjectListItem[] }>({
   178→    queryKey: ["projects"],
   179→    queryFn: async () => {
   180→      const res = await fetch("/api/v1/projects");
   181→      if (!res.ok) return { data: [] };
   182→      return res.json();
   183→    },
   184→  });
   185→  const projects = projectsData?.data ?? [];
   186→
   187→  const { data: projectDetail, isLoading: loadingDetail } = useQuery<{ data: ProjectDetail }>({
   188→    queryKey: ["project", selectedProjectId],
   189→    queryFn: async () => {
   190→      const res = await fetch(`/api/v1/projects/${selectedProjectId}`);
   191→      if (!res.ok) throw new Error("Failed to load project");
   192→      return res.json();
   193→    },
   194→    enabled: !!selectedProjectId,
   195→  });
   196→
   197→  // ------------------------- Mutations -------------------------
   198→  const createProjectMutation = useMutation({
   199→    mutationFn: async (input: { name: string; description?: string }) => {
   200→      const res = await fetch("/api/v1/projects", {
   201→        method: "POST",
   202→        headers: { "Content-Type": "application/json" },
   203→        body: JSON.stringify(input),
   204→      });
   205→      const json = await res.json();
   206→      if (!res.ok) {
   207→        throw new Error(json?.error?.message ?? "Failed to create project");
   208→      }
   209→      return json as { data: ProjectListItem; message: string };
   210→    },
   211→    onSuccess: (data) => {
   212→      queryClient.invalidateQueries({ queryKey: ["projects"] });
   213→      setShowNewProjectForm(false);
   214→      setSelectedProjectId(data.data.id);
   215→      toast.success("Project created", { description: data.data.name });
   216→    },
   217→    onError: (err: Error) => toast.error(err.message),
   218→  });
   219→
   220→  const updateProjectMutation = useMutation({
   221→    mutationFn: async (input: {
   222→      id: string;
   223→      name?: string;
   224→      description?: string | null;
   225→    }) => {
   226→      const res = await fetch(`/api/v1/projects/${input.id}`, {
   227→        method: "PATCH",
   228→        headers: { "Content-Type": "application/json" },
   229→        body: JSON.stringify({
   230→          name: input.name,
   231→          description: input.description ?? null,
   232→        }),
   233→      });
   234→      const json = await res.json();
   235→      if (!res.ok) {
   236→        throw new Error(json?.error?.message ?? "Failed to update project");
   237→      }
   238→      return json as { data: ProjectDetail };
   239→    },
   240→    onSuccess: () => {
   241→      queryClient.invalidateQueries({ queryKey: ["projects"] });
   242→      queryClient.invalidateQueries({ queryKey: ["project", selectedProjectId] });
   243→      toast.success("Project updated");
   244→    },
   245→    onError: (err: Error) => toast.error(err.message),
   246→  });
   247→
   248→  const deleteProjectMutation = useMutation({
   249→    mutationFn: async (id: string) => {
   250→      const res = await fetch(`/api/v1/projects/${id}`, { method: "DELETE" });
   251→      const json = await res.json();
   252→      if (!res.ok) {
   253→        throw new Error(json?.error?.message ?? "Failed to delete project");
   254→      }
   255→      return json;
   256→    },
   257→    onSuccess: (_data, deletedId) => {
   258→      if (selectedProjectId === deletedId) setSelectedProjectId(null);
   259→      queryClient.invalidateQueries({ queryKey: ["projects"] });
   260→      queryClient.invalidateQueries({ queryKey: ["project", deletedId] });
   261→      toast.success("Project deleted");
   262→    },
   263→    onError: (err: Error) => toast.error(err.message),
   264→  });
   265→
   266→  const addNoteMutation = useMutation({
   267→    mutationFn: async ({ projectId, content }: { projectId: string; content: string }) => {
   268→      const res = await fetch(`/api/v1/projects/${projectId}/notes`, {
   269→        method: "POST",
   270→        headers: { "Content-Type": "application/json" },
   271→        body: JSON.stringify({ content }),
   272→      });
   273→      const json = await res.json();
   274→      if (!res.ok) {
   275→        throw new Error(json?.error?.message ?? "Failed to add note");
   276→      }
   277→      return json;
   278→    },
   279→    onSuccess: () => {
   280→      queryClient.invalidateQueries({ queryKey: ["project", selectedProjectId] });
   281→      queryClient.invalidateQueries({ queryKey: ["projects"] });
   282→    },
   283→    onError: (err: Error) => toast.error(err.message),
   284→  });
   285→
   286→  // ------------------------- Render -------------------------
   287→  return (
   288→    <motion.div
   289→      initial={{ opacity: 0 }}
   290→      animate={{ opacity: 1 }}
   291→      exit={{ opacity: 0 }}
   292→      transition={{ duration: 0.2 }}
   293→      className="fixed inset-0 z-[60] flex bg-background/95 backdrop-blur-sm"
   294→      role="dialog"
   295→      aria-modal="true"
   296→      aria-label="Creative projects"
   297→    >
   298→      {/* Sidebar (desktop) */}
   299→      <aside className="hidden w-80 shrink-0 flex-col border-r border-border bg-surface/60 lg:flex">
   300→        <SidebarHeader onClose={onClose} />
   301→        <div className="border-b border-border p-3">
   302→          <Button
   303→            className="w-full justify-start bg-primary/15 text-primary hover:bg-primary/25"
   304→            variant="ghost"
   305→            onClick={() => {
   306→              setSelectedProjectId(null);
   307→              setShowNewProjectForm(true);
   308→            }}
   309→          >
   310→            <FolderPlus className="mr-2 h-4 w-4" />
   311→            New project
   312→          </Button>
   313→        </div>
   314→        <ProjectList
   315→          projects={projects}
   316→          loading={loadingList}
   317→          selectedId={selectedProjectId}
   318→          onSelect={(id) => {
   319→            setSelectedProjectId(id);
   320→            setShowNewProjectForm(false);
   321→          }}
   322→          onDelete={(id) => deleteProjectMutation.mutate(id)}
   323→          deletingId={deleteProjectMutation.isPending ? selectedProjectId ?? "" : ""}
   324→        />
   325→      </aside>
   326→
   327→      {/* Main area */}
   328→      <div className="flex flex-1 flex-col min-w-0">
   329→        {/* Mobile header */}
   330→        <header className="flex items-center justify-between border-b border-border px-4 py-3 lg:hidden">
   331→          <div className="flex items-center gap-2">
   332→            {selectedProjectId ? (
   333→              <Button
   334→                size="icon"
   335→                variant="ghost"
   336→                className="h-8 w-8"
   337→                onClick={() => {
   338→                  setSelectedProjectId(null);
   339→                  setMobileSidebarOpen(true);
   340→                }}
   341→                aria-label="Back to projects"
   342→              >
   343→                <ArrowLeft className="h-4 w-4" />
   344→              </Button>
   345→            ) : (
   346→              <WalloraMonogram className="h-7 w-7" />
   347→            )}
   348→            <span className="font-display text-sm font-semibold">Projects</span>
   349→          </div>
   350→          <Button
   351→            size="icon"
   352→            variant="ghost"
   353→            className="h-8 w-8"
   354→            onClick={onClose}
   355→            aria-label="Close projects panel"
   356→          >
   357→            <X className="h-4 w-4" />
   358→          </Button>
   359→        </header>
   360→
   361→        {/* Mobile project picker (Sheet-style overlay) */}
   362→        <AnimatePresence>
   363→          {mobileSidebarOpen && !selectedProjectId && (
   364→            <motion.div
   365→              initial={{ opacity: 0 }}
   366→              animate={{ opacity: 1 }}
   367→              exit={{ opacity: 0 }}
   368→              className="absolute inset-0 z-10 flex flex-col bg-background lg:hidden"
   369→            >
   370→              <div className="flex items-center justify-between border-b border-border px-4 py-3">
   371→                <span className="font-display text-sm font-semibold">Your projects</span>
   372→                <Button
   373→                  size="icon"
   374→                  variant="ghost"
   375→                  className="h-8 w-8"
   376→                  onClick={() => setMobileSidebarOpen(false)}
   377→                  aria-label="Close project list"
   378→                >
   379→                  <X className="h-4 w-4" />
   380→                </Button>
   381→              </div>
   382→              <div className="border-b border-border p-3">
   383→                <Button
   384→                  className="w-full justify-start bg-primary/15 text-primary hover:bg-primary/25"
   385→                  variant="ghost"
   386→                  onClick={() => {
   387→                    setShowNewProjectForm(true);
   388→                    setMobileSidebarOpen(false);
   389→                  }}
   390→                >
   391→                  <FolderPlus className="mr-2 h-4 w-4" />
   392→                  New project
   393→                </Button>
   394→              </div>
   395→              <ProjectList
   396→                projects={projects}
   397→                loading={loadingList}
   398→                selectedId={selectedProjectId}
   399→                onSelect={(id) => {
   400→                  setSelectedProjectId(id);
   401→                  setMobileSidebarOpen(false);
   402→                  setShowNewProjectForm(false);
   403→                }}
   404→                onDelete={(id) => deleteProjectMutation.mutate(id)}
   405→                deletingId=""
   406→              />
   407→            </motion.div>
   408→          )}
   409→        </AnimatePresence>
   410→
   411→        {/* Desktop header */}
   412→        <header className="hidden items-center justify-between border-b border-border px-6 py-4 lg:flex">
   413→          <div className="flex items-center gap-3">
   414→            <span className="flex h-9 w-9 items-center justify-center rounded-xl border border-border bg-surface text-primary">
   415→              <FolderKanban className="h-4.5 w-4.5" />
   416→            </span>
   417→            <div>
   418→              <h2 className="font-display text-base font-semibold text-foreground">
   419→                Creative Projects
   420→              </h2>
   421→              <p className="text-xs text-secondary-text">
   422→                Curate wallpapers + generations into private collections.
   423→              </p>
   424→            </div>
   425→          </div>
   426→          <Button
   427→            size="icon"
   428→            variant="ghost"
   429→            className="h-8 w-8"
   430→            onClick={onClose}
   431→            aria-label="Close projects panel"
   432→          >
   433→            <X className="h-4 w-4" />
   434→          </Button>
   435→        </header>
   436→
   437→        {/* Body */}
   438→        <div className="flex-1 overflow-y-auto">
   439→          <div className="mx-auto max-w-5xl p-4 sm:p-6">
   440→            <AnimatePresence mode="wait">
   441→              {showNewProjectForm && !selectedProjectId ? (
   442→                <motion.div
   443→                  key="new-project"
   444→                  initial={{ opacity: 0, y: 12 }}
   445→                  animate={{ opacity: 1, y: 0 }}
   446→                  exit={{ opacity: 0, y: -12 }}
   447→                  transition={{ duration: 0.2 }}
   448→                >
   449→                  <NewProjectForm
   450→                    onCancel={() => setShowNewProjectForm(false)}
   451→                    onSubmit={(input) => createProjectMutation.mutate(input)}
   452→                    isSubmitting={createProjectMutation.isPending}
   453→                  />
   454→                </motion.div>
   455→              ) : selectedProjectId && projectDetail?.data ? (
   456→                <motion.div
   457→                  key={`project-${selectedProjectId}`}
   458→                  initial={{ opacity: 0, y: 12 }}
   459→                  animate={{ opacity: 1, y: 0 }}
   460→                  exit={{ opacity: 0, y: -12 }}
   461→                  transition={{ duration: 0.2 }}
   462→                >
   463→                  <ProjectDetail
   464→                    project={projectDetail.data}
   465→                    onBack={() => setSelectedProjectId(null)}
   466→                    onUpdate={(input) =>
   467→                      updateProjectMutation.mutate({ id: selectedProjectId, ...input })
   468→                    }
   469→                    isUpdating={updateProjectMutation.isPending}
   470→                    onDelete={() => deleteProjectMutation.mutate(selectedProjectId)}
   471→                    isDeleting={deleteProjectMutation.isPending}
   472→                    onAddNote={(content) =>
   473→                      addNoteMutation.mutate({ projectId: selectedProjectId, content })
   474→                    }
   475→                    isAddingNote={addNoteMutation.isPending}
   476→                  />
   477→                </motion.div>
   478→              ) : selectedProjectId && loadingDetail ? (
   479→                <motion.div
   480→                  key="loading"
   481→                  initial={{ opacity: 0 }}
   482→                  animate={{ opacity: 1 }}
   483→                  exit={{ opacity: 0 }}
   484→                  className="flex h-64 items-center justify-center"
   485→                >
   486→                  <Loader2 className="h-6 w-6 animate-spin text-primary" />
   487→                </motion.div>
   488→              ) : (
   489→                <motion.div
   490→                  key="empty"
   491→                  initial={{ opacity: 0, y: 12 }}
   492→                  animate={{ opacity: 1, y: 0 }}
   493→                  exit={{ opacity: 0, y: -12 }}
   494→                  transition={{ duration: 0.2 }}
   495→                >
   496→                  <EmptyState
   497→                    hasProjects={projects.length > 0}
   498→                    onNewProject={() => setShowNewProjectForm(true)}
   499→                    onOpenMobileList={() => setMobileSidebarOpen(true)}
   500→                  />
   501→                </motion.div>
   502→              )}
   503→            </AnimatePresence>
   504→          </div>
   505→        </div>
   506→      </div>
   507→    </motion.div>
   508→  );
   509→}

   535→function ProjectList({
   536→  projects,
   537→  loading,
   538→  selectedId,
   539→  onSelect,
   540→  onDelete,
   541→  deletingId,
   542→}: {
   543→  projects: ProjectListItem[];
   544→  loading: boolean;
   545→  selectedId: string | null;
   546→  onSelect: (id: string) => void;
   547→  onDelete: (id: string) => void;
   548→  deletingId: string;
   549→}) {
   550→  if (loading) {
   551→    return (
   552→      <div className="flex flex-1 items-center justify-center p-6">
   553→        <Loader2 className="h-5 w-5 animate-spin text-secondary-text" />
   554→      </div>
   555→    );
   556→  }
   557→
   558→  if (projects.length === 0) {
   559→    return (
   560→      <div className="flex flex-1 flex-col items-center gap-2 p-6 text-center">
   561→        <FolderKanban className="h-8 w-8 text-secondary-text" />
   562→        <p className="text-xs text-secondary-text">
   563→          No projects yet.
   564→          <br />
   565→          Create your first to get started.
   566→        </p>
   567→      </div>
   568→    );
   569→  }
   570→
   571→  return (
   572→    <div className="flex-1 overflow-y-auto p-2">
   573→      {projects.map((p) => (
   574→        <div
   575→          key={p.id}
   576→          className={cn(
   577→            "group mb-1 cursor-pointer rounded-lg px-3 py-2.5 transition-colors",
   578→            selectedId === p.id
   579→              ? "bg-raised text-foreground"
   580→              : "text-secondary-text hover:bg-surface hover:text-foreground",
   581→          )}
   582→          onClick={() => onSelect(p.id)}
   583→        >
   584→          <div className="flex items-center gap-2">
   585→            <span className="flex h-6 w-6 shrink-0 items-center justify-center rounded-md border border-border bg-surface text-primary">
   586→              {p.isPublic ? (
   587→                <Globe className="h-3 w-3" />
   588→              ) : (
   589→                <Lock className="h-3 w-3" />
   590→              )}
   591→            </span>
   592→            <span className="flex-1 truncate text-xs font-medium">{p.name}</span>
   593→            <button
   594→              type="button"
   595→              onClick={(e) => {
   596→                e.stopPropagation();
   597→                if (
   598→                  confirm(
   599→                    `Delete "${p.name}"? This removes the project and all its assets and notes. This cannot be undone.`,
   600→                  )
   601→                ) {
   602→                  onDelete(p.id);
   603→                }
   604→              }}
   605→              className="opacity-0 transition-opacity group-hover:opacity-100"
   606→              aria-label="Delete project"
   607→            >
   608→              {deletingId === p.id ? (
   609→                <Loader2 className="h-3 w-3 animate-spin text-destructive" />
   610→              ) : (
   611→                <Trash2 className="h-3 w-3 text-destructive" />
   612→              )}
   613→            </button>
   614→          </div>
   615→          <div className="mt-1 flex items-center gap-2 pl-8 text-[10px] text-secondary-text">
   616→            <span>{p.assetCount} assets</span>
   617→            <span aria-hidden>·</span>
   618→            <span>{p.noteCount} notes</span>
   619→          </div>
   620→        </div>
