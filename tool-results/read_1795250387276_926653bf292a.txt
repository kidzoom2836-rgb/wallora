   133→const ADMIN_ROLES = new Set(["admin", "super_admin"]);
   134→
   135→function isAdminRole(role?: string) {
   136→  return !!role && ADMIN_ROLES.has(role);
   137→}

   139→// Higher-level moderation role set used for read access on stats/users/wallpapers
   140→const MOD_ROLES = new Set(["moderator", "admin", "super_admin"]);

   142→// ---------------------------------------------------------------------------
   143→// Main component

   145→
   146→export function AdminPanel() {
   147→  const { open, setOpen } = useAdminPanel();
   148→  const { data: session, status } = useSession();
   149→  const [tab, setTab] = React.useState<TabId>("overview");
   150→
   151→  // Block rendering for non-admin sessions even if the store says open.
   152→  const role = session?.user?.role;
   153→  const canRead = status === "authenticated" && MOD_ROLES.has(role ?? "");
   154→  const canModerate = status === "authenticated" && isAdminRole(role);
   155→
   156→  // Lock body scroll while open.
   157→  React.useEffect(() => {
   158→    if (!open) return;
   159→    const prev = document.body.style.overflow;
   160→    document.body.style.overflow = "hidden";
   161→    const onKey = (e: KeyboardEvent) => {
   162→      if (e.key === "Escape") setOpen(false);
   163→    };
   164→    window.addEventListener("keydown", onKey);
   165→    return () => {
   166→      document.body.style.overflow = prev;
   167→      window.removeEventListener("keydown", onKey);
   168→    };
   169→  }, [open, setOpen]);
   170→
   171→  return (
   172→    <AnimatePresence>
   173→      {open && (
   174→        <motion.div
   175→          initial={{ opacity: 0 }}
   176→          animate={{ opacity: 1 }}
   177→          exit={{ opacity: 0 }}
   178→          transition={{ duration: 0.2 }}
   179→          className="fixed inset-0 z-[60] flex flex-col bg-background/97 backdrop-blur-md"
   180→          role="dialog"
   181→          aria-modal="true"
   182→          aria-label="WALLORA Admin Dashboard"
   183→        >
   184→          {/* Header */}
   185→          <header className="flex items-center justify-between gap-4 border-b border-border px-4 py-3 sm:px-6">
   186→            <div className="flex items-center gap-3">
   187→              <WalloraLogo spark />
   188→              <div className="hidden sm:block">
   189→                <div className="flex items-center gap-2">
   190→                  <h1 className="font-display text-sm font-semibold uppercase tracking-[0.18em] text-foreground">
   191→                    Admin Dashboard
   192→                  </h1>
   193→                  <Badge
   194→                    variant="outline"
   195→                    className="border-primary/30 bg-primary/10 text-[10px] font-medium text-primary"
   196→                  >
   197→                    Phase 12
   198→                  </Badge>
   199→                </div>
   200→                <p className="text-[11px] text-secondary-text">
   201→                  Internal moderation &amp; platform metrics
   202→                </p>
   203→              </div>
   204→            </div>
   205→
   206→            <div className="flex items-center gap-2">
   207→              {canRead && (
   208→                <Badge
   209→                  variant="outline"
   210→                  className="hidden border-border bg-surface text-[10px] capitalize text-secondary-text sm:inline-flex"
   211→                >
   212→                  <Shield className="mr-1 h-2.5 w-2.5" />
   213→                  {role}
   214→                </Badge>
   215→              )}
   216→              <Button
   217→                size="icon"
   218→                variant="ghost"
   219→                className="h-9 w-9"
   220→                onClick={() => setOpen(false)}
   221→                aria-label="Close admin dashboard"
   222→              >
   223→                <X className="h-4 w-4" />
   224→              </Button>
   225→            </div>
   226→          </header>
   227→
   228→          {/* Tab navigation */}
   229→          <nav
   230→            className="flex items-center gap-1 overflow-x-auto border-b border-border px-4 py-2 sm:px-6"
   231→            aria-label="Admin sections"
   232→          >
   233→            <TabButton active={tab === "overview"} onClick={() => setTab("overview")}>
   234→              <LayoutDashboard className="h-3.5 w-3.5" />
   235→              Overview
   236→            </TabButton>
   237→            <TabButton active={tab === "users"} onClick={() => setTab("users")}>
   238→              <UsersIcon className="h-3.5 w-3.5" />
   239→              Users
   240→            </TabButton>
   241→            <TabButton active={tab === "moderation"} onClick={() => setTab("moderation")}>
   242→              <Shield className="h-3.5 w-3.5" />
   243→              Moderation
   244→              <PendingBadge />
   245→            </TabButton>
   246→          </nav>
   247→
   248→          {/* Body */}
   249→          <div className="flex-1 overflow-y-auto">
   250→            <AnimatePresence mode="wait">
   251→              <motion.div
   252→                key={tab}
   253→                initial={{ opacity: 0, y: 6 }}
   254→                animate={{ opacity: 1, y: 0 }}
   255→                exit={{ opacity: 0, y: -6 }}
   256→                transition={{ duration: 0.18 }}
   257→                className="mx-auto w-full max-w-7xl px-4 py-6 sm:px-6 sm:py-8"
   258→              >
   259→                {!canRead ? (
   260→                  <AccessDenied />
   261→                ) : tab === "overview" ? (
   262→                  <OverviewTab />
   263→                ) : tab === "users" ? (
   264→                  <UsersTab />
   265→                ) : (
   266→                  <ModerationTab canModerate={canModerate} />
   267→                )}
   268→              </motion.div>
   269→            </AnimatePresence>
   270→          </div>
   271→        </motion.div>
   272→      )}
   273→    </AnimatePresence>
   274→  );
   275→}

   307→function PendingBadge() {
   308→  const { data } = useQuery<{ data: Stats }>({
   309→    queryKey: ["admin-stats"],
   310→    queryFn: async () => {
   311→      const res = await fetch("/api/v1/admin/stats");
   312→      if (!res.ok) throw new Error("Failed to load stats");
   313→      return res.json();
   314→    },
   315→    placeholderData: keepPreviousData,
   316→  });
   317→  const pending = data?.data?.pendingWallpapers ?? 0;
   318→  if (pending === 0) return null;
   319→  return (
   320→    <span className="ml-1 inline-flex h-4 min-w-[16px] items-center justify-center rounded-full bg-primary/20 px-1 text-[10px] font-semibold text-primary">
   321→      {pending}
   322→    </span>
   323→  );
   324→}

   347→function OverviewTab() {
   348→  const { data, isLoading, isError } = useQuery<{ data: Stats }>({
   349→    queryKey: ["admin-stats"],
   350→    queryFn: async () => {
   351→      const res = await fetch("/api/v1/admin/stats");
   352→      if (!res.ok) throw new Error("Failed to load stats");
   353→      return res.json();
   354→    },
   355→  });
   356→
   357→  if (isError) {
   358→    return (
   359→      <div className="flex flex-col items-center gap-4 py-16 text-center">

   373→
   374→  const cards: Array<{
   375→    label: string;

   378→    hint?: string;

   381→    { label: "Wallpapers", value: s.totalWallpapers, icon: <ImageIcon className="h-4 w-4" />, hint: `+${s.todayNewWallpapers} today` },
   382→    { label: "Pending review", value: s.pendingWallpapers, icon: <Clock className="h-4 w-4" /> },
   383→    { label: "AI conversations", value: s.totalConversations, icon: <MessageSquare className="h-4 w-4" /> },
   384→    { label: "AI messages", value: s.totalMessages, icon: <MessageCircle className="h-4 w-4" /> },
   385→    { label: "AI generations", value: s.totalGenerations, icon: <Sparkles className="h-4 w-4" /> },
   386→    { label: "Comments", value: s.totalComments, icon: <MessageCircle className="h-4 w-4" /> },
   387→    { label: "Products", value: s.totalProducts, icon: <ShoppingBag className="h-4 w-4" /> },
   388→    { label: "Purchases", value: s.totalPurchases, icon: <ShoppingCart className="h-4 w-4" /> },
   389→    { label: "Waitlist entries", value: s.totalWaitlistEntries, icon: <Mail className="h-4 w-4" /> },
   390→  ];

   392→  return (
   393→    <div className="space-y-6">
   394→      {/* Top KPI row */}

   402→            <StatCard key={c.label} {...c} />
   403→          ))}
   404→        </div>
   405→      </section>
   406→
   407→      {/* Today usage + status breakdown */}
   408→      <section className="grid gap-4 md:grid-cols-3">
   409→        <UsageCard credits={s.todayAiUsage} newUsers={s.todayNewUsers} newWallpapers={s.todayNewWallpapers} />

   473→function StatusBreakdownCard({
   474→  data,
   475→}: {
   476→  data: { approved: number; pending: number; rejected: number; flagged: number };
   477→}) {
   478→  const rows: Array<{ key: string; label: string; value: number; color: string }> = [
   479→    { key: "approved", label: "Approved", value: data.approved, color: "bg-success" },
   480→    { key: "pending", label: "Pending", value: data.pending, color: "bg-primary" },
   481→    { key: "rejected", label: "Rejected", value: data.rejected, color: "bg-destructive" },
   482→    { key: "flagged", label: "Flagged", value: data.flagged, color: "bg-secondary-text" },
   483→  ];
   484→  const total = rows.reduce((acc, r) => acc + r.value, 0) || 1;
   485→  return (
   486→    <div className="rounded-xl border border-border bg-surface p-5">
   487→      <div className="flex items-center gap-2">
   488→        <Shield className="h-4 w-4 text-primary" />
   489→        <h3 className="font-display text-sm font-semibold text-foreground">Wallpaper status</h3>
   490→      </div>
   491→      <div className="mt-3 flex h-2.5 overflow-hidden rounded-full bg-raised">
   492→        {rows.map((r) =>
   493→          r.value > 0 ? (
   494→            <div
   495→              key={r.key}
   496→              className={cn("h-full", r.color)}
   497→              style={{ width: `${(r.value / total) * 100}%` }}
   498→              title={`${r.label}: ${r.value}`}
   499→            />
   500→          ) : null,
   501→        )}
   502→      </div>
   503→      <dl className="mt-4 space-y-2.5">
   504→        {rows.map((r) => (
   505→          <div key={r.key} className="flex items-center justify-between text-xs">
   506→            <dt className="flex items-center gap-2 text-secondary-text">
   507→              <span className={cn("h-2 w-2 rounded-full", r.color)} />
   508→              {r.label}
   509→            </dt>
   510→            <dd className="tabular-nums font-medium text-foreground">
   511→              {r.value.toLocaleString()}
   512→            </dd>
   513→          </div>
   514→        ))}
   515→      </dl>
   516→    </div>
   517→  );
   518→}

   807→  const limit = 24;

   810→  // Reset to page 1 when filter changes.

   812→    setPage(1);

   814→

   816→    data: {
   817→      items: AdminWallpaper[];
   818→      pagination: { page: number; limit: number; total: number; totalPages: number };
   819→    };
   820→  }>({
   821→    queryKey: ["admin-wallpapers", status, page],
   822→    queryFn: async () => {
   823→      const res = await fetch(
   824→        `/api/v1/admin/wallpapers?status=${status}&page=${page}&limit=${limit}`,

   834→      const res = await fetch(`/api/v1/admin/wallpapers/${id}/moderate`, {
   835→        method: "PATCH",
   836→        headers: { "Content-Type": "application/json" },
   837→        body: JSON.stringify({ status: nextStatus }),
   838→      });

   841→        throw new Error(payload?.error?.message ?? "Moderation failed");

   843→      return payload;
   844→    },

   847→        description: "The moderation decision has been recorded.",

   850→      queryClient.invalidateQueries({ queryKey: ["admin-stats"] });

   852→    onError: (err: Error) => {
   853→      toast.error("Moderation failed", { description: err.message });
   854→    },

   858→  const pagination = data?.data.pagination;

   875→                status === opt.id
   876→                  ? "bg-raised text-foreground"
   877→                  : "text-secondary-text hover:text-foreground",
   878→              )}
   879→            >
   880→              {opt.label}
   881→            </button>
   882→          ))}
   883→        </div>
   884→      </div>
   885→
   886→      {!canModerate && (
   887→        <div className="flex items-center gap-2 rounded-lg border border-border bg-surface px-4 py-3 text-xs text-secondary-text">
   888→          <Lock className="h-3.5 w-3.5" />
   889→          You have read access only. Approve and reject actions require an admin role.
   890→        </div>
   891→      )}
   892→
   893→      {isError ? (
   894→        <div className="flex flex-col items-center gap-4 py-16 text-center">
   895→          <span className="flex h-12 w-12 items-center justify-center rounded-full bg-destructive/10 text-destructive">

   901→        <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
   902→          {Array.from({ length: 8 }).map((_, i) => (
   903→            <div key={i} className="aspect-[4/5] animate-pulse rounded-xl border border-border bg-surface" />
   904→          ))}
   905→        </div>
   906→      ) : items.length === 0 ? (
   907→        <EmptyState
   908→          icon={<CheckCircle2 className="h-7 w-7" />}

   913→        <>
   914→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">

   916→              <ModerationCard
   917→                key={w.id}

   919→                canModerate={canModerate}
   920→                pendingAction={moderateMutation.isPending}
   921→                onApprove={() => moderateMutation.mutate({ id: w.id, nextStatus: "approved" })}
   922→                onReject={() => moderateMutation.mutate({ id: w.id, nextStatus: "rejected" })}
   923→                onFlag={() => moderateMutation.mutate({ id: w.id, nextStatus: "flagged" })}
   924→              />
   925→            ))}
   926→          </div>
   927→          {pagination && pagination.totalPages > 1 && (
   928→            <Pagination
   929→              page={pagination.page}
   930→              totalPages={pagination.totalPages}
